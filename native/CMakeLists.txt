cmake_minimum_required(VERSION 3.15)
project(PokemonOcgCore LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS "ocgcore/*.cpp" "ocgcore/RNG/*.cpp")
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "ocgcore/lua/src/*.c")
foreach(EXCLUDED lbitlib lcorolib ldblib linit loadlib loslib ltests lua luac lutf8lib onelua)
  list(FILTER LUA_SOURCES EXCLUDE REGEX "/${EXCLUDED}\\.c$")
endforeach()
set_source_files_properties(${LUA_SOURCES} PROPERTIES LANGUAGE CXX)
add_library(ocgcore SHARED ${CORE_SOURCES} ${LUA_SOURCES})
target_compile_features(ocgcore PRIVATE cxx_std_17)
target_include_directories(ocgcore PRIVATE ocgcore/lua ocgcore/lua/src)
target_compile_definitions(ocgcore PRIVATE OCGCORE_EXPORT_FUNCTIONS WIN32 _WIN32 NOMINMAX NDEBUG)
if(MSVC)
  target_compile_options(ocgcore PRIVATE /GR- /wd4244 /FIluaconf-customize.h)
endif()
